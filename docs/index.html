Plug into DI

Register validators from your assembly and inject the typed validator where it belongs.

UserValidator.cs 
using Validtron;
using Validtron.Extensions;

public sealed class UserValidator : Validator<User>
{
    public UserValidator()
    {
        RuleFor(x => x.Email)
            .NotEmpty()
            .EmailAddress();

        RuleFor(x => x.Age)
            .GreaterThanOrEqual(18);
    }
}
var validator = new UserValidator();
var result = validator.Validate(user);

if (!result.IsValid)
{
    foreach (var error in result.Errors)
    {
        Console.WriteLine(
            $"{error.PropertyName}: {error.ErrorMessage}");
    }
}
using Validtron.DependencyInjection;

services.AddValidtron(
    typeof(Program).Assembly);

// Resolve IValidator<User> through DI
public sealed class UserService(
    IValidator<User> validator)
{
    // ...
}
Async when you need it

Keep external checks inside the validation flow.

Use asynchronous predicates for uniqueness checks, database-backed rules or service calls, then run the validator with ValidateAsync.

See async validation → 
RuleFor(x => x.UserName)
    .NotEmpty()
    .MustAsync(async (value, ct) =>
        !await userStore.ExistsAsync(value!, ct));

var result = await validator.ValidateAsync(model);
